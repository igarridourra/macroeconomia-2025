\item \textbf{Sol.:}
\begin{align*}
\text{PIB nominal} &= \sum (\text{precio actual}\times \text{cantidad actual})\\
\text{2017: } & 100\cdot100 + 200\cdot50 = 20{,}000\\
\text{2018: } & 200\cdot150 + 300\cdot100 = 60{,}000\\
\text{2019: } & 300\cdot200 + 400\cdot150 = 120{,}000
\end{align*}
\[
\text{PIB real (base 2017)}=\sum (\text{precio 2017}\times \text{cantidad actual})
\]
\begin{align*}
\text{2017: } & 100\cdot100 + 200\cdot50 = 20{,}000\\
\text{2018: } & 100\cdot150 + 200\cdot100 = 35{,}000\\
\text{2019: } & 100\cdot200 + 200\cdot150 = 50{,}000
\end{align*}
\[
\text{Deflactor del PIB} = \frac{\text{PIB nominal}}{\text{PIB real}}\times 100
\]
\begin{align*}
2017:&\; (20{,}000/20{,}000)\times 100 = 100\%\\
2018:&\; (60{,}000/35{,}000)\times 100 = 171{,}42\%\\
2019:&\; (120{,}000/50{,}000)\times 100 = 240\%
\end{align*}
\textbf{IPC (base 2017):} \; IPC$_{2017}=100\%$, \; IPC$_{2018}=150\%$, \; IPC$_{2019}=300\%$.\\
\textbf{Inflación (vía IPC):}\; 2018: $150\%-100\%=50\%$;\; 2019: $300\%-150\%=150\%$.
