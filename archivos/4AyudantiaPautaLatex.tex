% ========================= 2024 - P1 =========================
\subsection*{(Certamen 1, 2024) \quad Pregunta 1 (1,5 puntos)}
\textbf{Enunciado.} Curva de Phillips con expectativas:
\[
\pi_t = \pi_{t-1} + \big(0{,}05 - 1{,}25\,u_t\big).
\]

\textbf{Pauta / Solución.}

\begin{enumerate}[label=\alph*)]
\item \textbf{Expectativas racionales.} No. La forma \(\pi_t = \pi_{t-1} + \cdots\) implica expectativas \emph{adaptativas} (\(\pi_t^{e}=\pi_{t-1}\)), no racionales. Bajo expectativas racionales, \(\pi_t^{e}\) sería la esperanza matemática condicional, no necesariamente \(\pi_{t-1}\).

\item \textbf{Tasa natural de desempleo.} En equilibrio de inflación constante (\(\pi_t=\pi_{t-1}\)) se cumple:
\[
0 = 0{,}05 - 1{,}25\,u^n \quad \Rightarrow \quad u^n = \frac{0{,}05}{1{,}25} = 0{,}04 = \boxed{4\%}.
\]
\end{enumerate}


% ========================= 2024 - P2 =========================
\subsection*{(Certamen 1, 2024) \quad Pregunta 2 (1,5 puntos)}
\textbf{Enunciado.} OA: \(P_{t+1}=P_t\,[1+\lambda(Y-Y^*)]\). Analice y \emph{grafique} el ajuste ante una \textbf{fuerte caída de importaciones}.

\textbf{Pauta / Solución (idea económica y secuencia temporal).}
\begin{itemize}
\item \textbf{Impacto inicial (t=0).} Una caída exógena de importaciones (\(M\downarrow\)) eleva \(XN=X-M\) y, por identidad, desplaza la \textbf{DA} a la \emph{derecha}. Resultado de corto plazo: \(Y_0>Y^*\) y \(P_0\) comienza a subir.
\item \textbf{Dinámica (t=1,2,\dots).} Con \(Y>Y^*\), la ecuación de OA implica \(P_{t+1}>P_t\). El alza de \(P\) reduce saldos reales y \emph{modera} DA; simultáneamente, la OA \emph{desliza} la economía hacia \(Y^*\).
\item \textbf{Mediano plazo.} Se retorna a \(\boxed{Y=Y^*}\) con un \(\boxed{P}\) \textbf{más alto}. La velocidad depende de \(\lambda\) (rigideces nominales).

\end{itemize}

\textbf{Gráfico sugerido.} Dos paneles OA--DA: (i) choque DA\(\uparrow\) (de \(DA_0\) a \(DA_1\)), (ii) senda temporal \(P_t\) al alza con \(Y_t\to Y^*\).

\newpage
\vspace*{0.1cm}
\textbf{En terminos graficos:}

\textbf{OA: } \(P_{t+1}=P_t[1+\lambda(Y-Y^*)]\). \textbf{Choque:} fuerte caída de importaciones (\(M\downarrow\)).

\textbf{Idea económica.} \(DA\) se desplaza \(\rightarrow\) por \(XN\uparrow\). En el corto plazo \(Y>Y^*\) y \(P\) sube; luego la dinámica de \(P\) (vía \(\lambda\)) va llevando \(Y\) a \(Y^*\) con \(P\) más alto.

\subsection*{(Certamen 1, 2024) \quad Pregunta 4 (1,5 puntos)}
\textbf{Enunciado.} Aumento de \(\pi^e\) (\(+0{,}4\) pp) por descongelamiento de tarifas eléctricas. OA--DA con \textbf{expectativas racionales}, sin política y con \emph{consumo agregado constante} (hogares sustituyen).

\textbf{Pauta / Solución.}
\begin{itemize}
\item \textbf{DA:} Se mantiene (consumo total constante, \(G\) e \(I\) sin cambios).
\item \textbf{OA de corto plazo:} Con expectativas \emph{racionales}, el alza anticipada de precios/tarifas se incorpora en salarios y precios fijados, \(\Rightarrow\) \textbf{OA} se desplaza \emph{hacia arriba} \emph{de inmediato}.

\end{itemize}

\textbf{Choque de expectativas racionales: } alza anticipada de tarifas eléctricas \(\Rightarrow \pi^e\uparrow\).

\textbf{Resultado.} Con expectativas \emph{racionales} y sin política, la \textbf{OA} de corto plazo sube \emph{de inmediato}; \(DA\) estable. Se salta a \(P\) más alto con \(Y=Y^*\) (desempleo en la tasa natural).

\textbf{Pauta / Solución.}

\[
\text{Alimentos: } 25\cdot 100 = 2500;\quad
\text{Vestuario: } 40\cdot 50 = 2000;\quad
\text{Energía: } 10\cdot 150 = 1500.
\]
\[
\text{VBP} = 2500 + 2000 + 1500 = \boxed{6000}.
\]

\textit{Intermedios de Energía:} \(50\) a Alimentos (\(=500\)), \(50\) a Vestuario (\(=500\)) \(\Rightarrow\) \(\text{IC}=1000\).

\[
\text{PIB (VA)} = \text{VBP} - \text{IC} = 6000 - 1000 = \boxed{5000}.
\]

\textit{Comercio exterior (CLP):}
\[
X = 50 \times US\$1{,}25 \times 20 = \boxed{1250}, \quad
M = 10 \times US\$1{,}5 \times 20 = \boxed{300},
\]
\[
XN = X - M = \boxed{950}.
\]

\newpage
\vspace*{0.1cm}
\textit{Consumo de hogares (incluye importado):}
\[
C = \underbrace{50\times 25}_{\text{Alim. }=1250} + \underbrace{50\times 40}_{\text{Vest. }=2000} + \underbrace{50\times 10}_{\text{Energ. }=500} + \underbrace{300}_{\text{M}} = \boxed{4050}.

\textit{Chequeo por gasto:} \(PIB = C + I + G + (X-M) = 4050 + 0 + 0 + 950 = \boxed{5000}.\)

\textbf{Datos.} Curva de Phillips con expectativas:
\[
\pi_t=\pi_t^{e}-0{,}5\,(u_t-0{,}06)
\]
Ley de Okun:
\[
\frac{Y_t-Y_t^*}{Y_t^*}=0{,}09-1{,}5\,u_t
\]
Se desea llevar la inflación a \(3\%\) este año; el año anterior fue \(5\%\), por lo que
\(\pi_t^{e}=\pi_{t-1}=5\%\) (expectativas adaptativas).

\begin{enumerate}[label=\alph*)]
\item \textbf{Tasa natural de desempleo.}\\
En inflación constante (\(\pi_t=\pi_t^{e}\)):
\[
\pi_t-\pi_t^{e}= -0{,}5\,(u^n-0{,}06)=0 \;\Rightarrow\; u^n=\boxed{0{,}06=6\%}.
\]

\item \textbf{Aumento necesario del desempleo (p.p.).}\\
Objetivo \(\pi_t=3\%\) con \(\pi_t^{e}=5\%\):
\[
0{,}03=0{,}05-0{,}5\,(u_t-0{,}06)
\;\Rightarrow\;
-0{,}02=-0{,}5u_t+0{,}03
\;\Rightarrow\;
-0{,}05=-0{,}5u_t
\;\Rightarrow\;
u_t=\boxed{0{,}10=10\%}.
\]
El desempleo debe subir en \(\boxed{10\%-6\%=4\text{ p.p.}}\).

\item \textbf{Variación del PIB respecto del potencial al lograr \(3\%\).}\\
Con \(u_t=10\%\) en Okun:
\[
\frac{Y_t-Y_t^*}{Y_t^*}=0{,}09-1{,}5(0{,}10)=0{,}09-0{,}15=\boxed{-0{,}06}
\]
\(\Rightarrow\) el PIB quedaría \(\boxed{6\%\ \text{por debajo del potencial}}.\)
\end{enumerate}

\subsection*{(Certamen 1, 2022) \quad Pregunta 2 (1,5 puntos)}

\textbf{Enunciado.}  
Si los agentes económicos tienen expectativas racionales y confían en que el Banco Central está comprometido con reducir la inflación, el costo de reducirla es mucho menor que cuando los agentes son escépticos.  

\begin{enumerate}[label=\alph*)]
\item Argumente si la afirmación es verdadera o falsa.  

\textbf{Pauta / Solución:}  
La afirmación es \textbf{verdadera}.  
La curva de Phillips con expectativas racionales indica que la inflación actual depende de la inflación esperada. Si los agentes confían en que el Banco Central cumplirá con la meta de inflación, ajustan sus expectativas de inmediato y no es necesario un aumento significativo del desempleo para bajar la inflación.  
En cambio, si los agentes son escépticos, la inflación esperada se mantiene alta, y el Banco Central debe generar una brecha de producto negativa mayor (más desempleo) para reducir la inflación efectiva. Por lo tanto, el costo de desinflar es mucho más alto bajo desconfianza.

\end{enumerate}

\subsection*{(Certamen 1, 2021) \quad Pregunta 2 (1,5 puntos)}

\textbf{Datos.} Población en edad de trabajar: \(15{,}92\) millones. Fuerza de trabajo (ocupados + desocupados): \(9{,}11\) millones. Desocupados: \(0{,}77\) millones. Empleo informal: \(1{,}32\) millones (subconjunto de los ocupados). Inactivos disponibles: \(0{,}91\) millones (no buscan, pero aceptarían empleo).

\begin{enumerate}[label=\alph*)]

\item \textbf{¿Cuál es el número de ocupados?}

Por definición, \(\text{Ocupados} = \text{Fuerza de trabajo} - \text{Desocupados}\).
\[
\text{Ocupados} = 9{,}11 - 0{,}77 = \boxed{8{,}34\ \text{millones}}
\]

\item \textbf{¿Cuál es la tasa de participación laboral (TPL)?}

\(\displaystyle \text{TPL} = \frac{\text{Fuerza de trabajo}}{\text{Población en edad de trabajar}}\times 100\).
\[
\text{TPL} = \frac{9{,}11}{15{,}92}\times 100 = \boxed{57{,}2\%}
\]

\item \textbf{¿Cuál es la tasa de desempleo (TD)?}

\(\displaystyle \text{TD} = \frac{\text{Desocupados}}{\text{Fuerza de trabajo}}\times 100\).
\[
\text{TD} = \frac{0{,}77}{9{,}11}\times 100 = \boxed{8{,}4\%}
\]

\end{enumerate}

\textit{Nota.} Los \emph{inactivos disponibles} (\(0{,}91\) millones) \textbf{no} se incluyen en la fuerza de trabajo bajo la definición estándar; si se pidiera una “tasa de desempleo ampliada”, se suman al numerador y al denominador.
