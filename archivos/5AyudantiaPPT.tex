% ===================== Resumen 1: Laboral =====================
\begin{frame}{Resumen — Mercado laboral: definiciones y fórmulas}
\textbf{Población en Edad de Trabajar (PET)}: personas $\ge$ 15 años. \\
\textbf{Fuerza Laboral (FL)} $=$ Ocupados $(O)$ $+$ Desocupados $(D)$.

\medskip
\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\begin{itemize}
  \item \textbf{Tasa de participación}:
  \[
  TP = \frac{FL}{PET}
  \]
  \item \textbf{Tasa de desempleo}:
  \[
  TD = \frac{D}{FL}
  \]
  \item \textbf{Criterio de desempleado (ENE/INE):} no trabajó la semana pasada, buscó empleo en últimas 4 semanas y está disponible 2 semanas.
\end{itemize}

\column{0.45\textwidth}
\textbf{Tipos de desempleo}
\begin{itemize}
  \item \textit{Natural (friccional/estructural)}: persiste incluso en largo plazo.
  \item \textit{Cíclico}: fluctuaciones alrededor del natural (ciclo económico).
\end{itemize}
\textbf{Ley de Okun} (idea): $\uparrow$ desempleo $\Rightarrow \downarrow$ PIB (pérdida de producto).
\end{columns}
\end{frame}

% ===================== Resumen 2: Ingreso-Gasto =====================
\begin{frame}{Resumen — Modelo ingreso–gasto (economía cerrada)}
\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\textbf{Identidades básicas}
\[
Y = C + I + G,\qquad \Yd = Y - T + TR
\]
\[
C = C_0 + c\,\Yd,\quad 0<c<1
\]

\textbf{Impuesto proporcional:}\; $T = tY$ \quad (si es fijo, $T=\bar T$)

\column{0.45\textwidth}
\textbf{Multiplicadores}
\[
k = \frac{1}{1-c}\quad (\text{sin }t)
\]
\[
k = \frac{1}{1 - c(1-t)}\quad (\text{con }t)
\]
\textbf{Balance fiscal}
\[
BF = T - G - TR
\]
\end{columns}
\end{frame}

% ===================== Resumen 3: Dinero / Inflación =====================
\begin{frame}{Resumen — Dinero e inflación (ideas clave)}
\begin{itemize}
  \item \textbf{Funciones del dinero:} medio de pago, unidad de cuenta, depósito de valor, liquidez.
  \item \textbf{Creación de dinero:} sistema de reserva fraccional $\Rightarrow$ multiplicador monetario $=1/R$.
  \item \textbf{Inflación} (costos): redistribución no anticipada, “costos de menú” y “suela de zapato”.
  \item \textbf{Indexación (UF en Chile):} reduce riesgo inflacionario en contratos de largo plazo.
\end{itemize}
\end{frame}
